The reset handle of the nearest [`ReadErrorResetBoundary`](/docs/reference/react/components#readerrorresetboundary), or of the whole app without
one. Call `reset` from an error boundary's reset callback so `useSuspenseRead` reads again.

## Signature

```ts
function useReadErrorReset(): ReadErrorReset
```

## Examples

```tsx title="Resetting from a boundary component"
import { useReadErrorReset } from "@ensforge/react"

export function RetryButton({ onRetry }: { onRetry: () => void }) {
  const { reset } = useReadErrorReset()
  return <button onClick={() => { reset(); onRetry() }}>Try again</button>
}
```
